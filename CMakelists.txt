cmake_minimum_required(VERSION 4.0)

project(electin
    VERSION 1.0.0
    DESCRIPTION "An ECS game engine in C++"
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Vortex config generator--
set(JSON_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/el_settings.json")
set(PYTHON_GENERATOR "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_config.py")
set(GENERATED_HPP_FILE "${CMAKE_CURRENT_SOURCE_DIR}/include/el/core/config/el_config.hpp")

add_custom_command(
    OUTPUT ${GENERATED_HPP_FILE}
    COMMAND python ${PYTHON_GENERATOR} ${JSON_CONFIG_FILE} ${GENERATED_HPP_FILE}
    DEPENDS ${JSON_CONFIG_FILE} ${PYTHON_GENERATOR}
    COMMENT "Generating el_generated_config.hpp from JSON..."
)

add_custom_target(GenerateConfig ALL DEPENDS ${GENERATED_HPP_FILE})

#----------------------------------------------------------------------------------------------------------------------------------

# THIRD PARTY LIBRARIES

# raylib---
add_subdirectory(vendor/raylib)
# Note: raylib's CMake automatically handles include directories and linking

# nlohmann/json---
# Since json is header-only, we create an INTERFACE library
add_library(json INTERFACE)
target_include_directories(json INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/json)

# odygrd/quill (The logger API)---
add_subdirectory(vendor/quill)

# erincatto/box2d---
add_subdirectory(vendor/box2d)

#----------------------------------------------------------------------------------------------------------------------------------

# INTERFACE for all the include paths
add_library(el_includes INTERFACE)
target_include_directories(el_includes INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include/")

#----------------------------------------------------------------------------------------------------------------------------------
# ENGINE SOURCE
add_subdirectory(src)

#----------------------------------------------------------------------------------------------------------------------------------
# EXECUTABLES / TESTS
if(PROJECT_IS_TOP_LEVEL)
    add_executable(el_test "${CMAKE_CURRENT_SOURCE_DIR}/test/test.cpp")
    target_link_libraries(el_test PRIVATE electin)
    add_dependencies(el_test GenerateConfig)
endif()
